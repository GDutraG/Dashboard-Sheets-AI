// ════════════════════════════════════════════════════════════════════ // DASHBOARD G & S — Apps Script v2 // ✅ Detecta meses automaticamente (sem hardcode de colunas) // ✅ Detecta categorias dinamicamente // ✅ Compatível com novas abas: Reserva, Investimentos, Sonhos // // INSTRUÇÕES: // 1. Extensões → Apps Script → cole este código → salve // 2. Implantar → Nova implantação → App da Web // - Executar como: Eu // - Acesso: Qualquer pessoa // 3. Copie a URL e adicione: ?key=gs2025financas // ════════════════════════════════════════════════════════════════════ // ── CONFIGURE AQUI ─────────────────────────────────────────────────── const SECRET_KEY = "gs2025financas"; // ← troque se quiser const SHEET_NAME = "Finanças"; // Linha onde ficam os cabeçalhos dos meses (linha 3 = Mai, Jun... Jan, Fev...) const ROW_MESES = 3; // Coluna onde os meses começam (G = 7) const COL_INICIO = 7; // Mês e ano do PRIMEIRO mês da planilha const MES_INICIO = "Mai"; const ANO_INICIO = 2025; // Labels buscados dinamicamente na coluna B const LABELS = { totalReceitas: "Total Receitas", totalDespesas: "Total Despesas", saldo: "SALDO TOTAL", }; // Categorias de despesa — nomes EXATOS como aparecem na coluna B // Cada categoria é a linha de subtotal (ex: "Moradia" soma aluguel, condomínio, etc.) const CATEGORIAS = [ "Moradia", "Alimentação", "Transporte", "Seguros", "Saúde", "Educação", "Lazer", "Cuidados Pessoais", "Vesturário", "Mensalidades", "Impostos/Taxas", "Outros", ]; // ── HANDLER PRINCIPAL ───────────────────────────────────────────────── function doGet(e) { // Verifica chave de segurança if (!e || e.parameter.key !== SECRET_KEY) { return jsonResponse({ erro: "Acesso negado" }); } // Rota: proxy de IA if (e.parameter.action === "ia") { return handleIA(e); } // Rota padrão: dados financeiros try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return jsonResponse({ erro: `Aba "${SHEET_NAME}" não encontrada` }); } // ── 1. Mapeia linhas pelos labels (busca na coluna B) ───────────── const linhas = encontrarLinhas(sheet, LABELS); // ── 2. Detecta meses disponíveis automaticamente ────────────────── // Meses ficam todos na linha 3: Mai Jun Jul Ago Set Out Nov Dez Jan Fev Mar... const lastCol = sheet.getLastColumn(); const totalCols = lastCol - COL_INICIO + 1; const mesRow = sheet.getRange(ROW_MESES, COL_INICIO, 1, totalCols).getValues()[0]; const mesesLabels = []; const mesesCols = []; const SEQ_MESES = ["Jan","Fev","Mar","Abr","Mai","Jun","Jul","Ago","Set","Out","Nov","Dez"]; const mesAbsInicio = ANO_INICIO * 12 + SEQ_MESES.indexOf(MES_INICIO); const ROW_TOTAL_REC = linhas.totalReceitas || 18; const ROW_TOTAL_DES = linhas.totalDespesas || 127; // Encontra âncora: primeira coluna com mês válido let primeiraColMes = -1; for (let i = 0; i < totalCols; i++) { if (SEQ_MESES.includes(String(mesRow[i]).trim())) { primeiraColMes = i; break; } } if (primeiraColMes === -1) return jsonResponse({ erro: "Nenhum mês encontrado na linha 3" }); let semDadosConsecutivos = 0; for (let i = primeiraColMes; i < totalCols; i++) { const mes = String(mesRow[i]).trim(); if (!SEQ_MESES.includes(mes)) break; const distancia = i - primeiraColMes; const mesAbs = mesAbsInicio + distancia; const anoCalc = Math.floor(mesAbs / 12); const mesEsperado = SEQ_MESES[mesAbs % 12]; if (mesEsperado !== mes) continue; // Inclui apenas meses com receita lançada (mês efetivamente aberto) // Meses com só despesas fixas (receita=0) são meses futuros pré-preenchidos const vRec = sheet.getRange(ROW_TOTAL_REC, COL_INICIO + i).getValue(); const vDes = sheet.getRange(ROW_TOTAL_DES, COL_INICIO + i).getValue(); const temReceita = vRec && vRec !== 0; const temDespesa = vDes && vDes !== 0; // Mês válido = tem receita OU tem despesa variável (não só fixas) // Paramos quando receita=0 por 2 meses consecutivos if (!temReceita) { semDadosConsecutivos++; if (semDadosConsecutivos >= 2) break; if (temDespesa) { // Inclui mesmo sem receita se for o primeiro mês sem (pode estar em preenchimento) mesesLabels.push(`${mes}/${String(anoCalc).slice(2)}`); mesesCols.push(COL_INICIO + i); } continue; } semDadosConsecutivos = 0; mesesLabels.push(`${mes}/${String(anoCalc).slice(2)}`); mesesCols.push(COL_INICIO + i); } const totalMeses = mesesCols.length; if (totalMeses === 0) { return jsonResponse({ erro: "Nenhum mês com dados encontrado" }); } // ── 3. Lê arrays de dados por linha ────────────────────────────── function lerLinha(numLinha) { if (!numLinha) return Array(totalMeses).fill(0); return mesesCols.map(col => { const v = sheet.getRange(numLinha, col).getValue(); return typeof v === "number" ? v : 0; }); } const receitas = lerLinha(linhas.totalReceitas); const despesas = lerLinha(linhas.totalDespesas); const saldo = lerLinha(linhas.saldo); // ── 4. Lê categorias dinamicamente ──────────────────────────────── const categorias = {}; CATEGORIAS.forEach(cat => { const linhaCat = encontrarLinha(sheet, cat); categorias[cat] = linhaCat ? lerLinha(linhaCat) : Array(totalMeses).fill(0); }); // ── 5. Lê dados das abas extras (se existirem) ──────────────────── const extras = {}; const sheetReserva = ss.getSheetByName("Reserva"); if (sheetReserva) { extras.reserva = { saldoAtual: lerCelula(sheetReserva, "C8"), meta6x: lerCelula(sheetReserva, "C6"), meta3x: lerCelula(sheetReserva, "C7"), mesesCobertos:lerCelula(sheetReserva, "C11"), situacao: lerCelula(sheetReserva, "C12"), }; } const sheetInvest = ss.getSheetByName("Investimentos"); if (sheetInvest) { extras.investimentos = { totalPatrimonio: lerCelula(sheetInvest, "C4"), totalRF: lerCelula(sheetInvest, "D31"), totalRV: lerCelula(sheetInvest, "D55"), rentabilidade: lerCelula(sheetInvest, "F4"), aporteTotal: lerCelula(sheetInvest, "H4"), }; } const sheetSonhos = ss.getSheetByName("Sonhos"); if (sheetSonhos) { extras.sonhos = lerSonhos(sheetSonhos); } // ── 6. Monta payload final ──────────────────────────────────────── const payload = { geradoEm: new Date().toISOString(), meses: mesesLabels, receitas, despesas, saldo, categorias, extras, }; return jsonResponse(payload); } catch (err) { return jsonResponse({ erro: "Erro interno: " + err.message }); } } // ── HELPERS ─────────────────────────────────────────────────────────── // Retorna JSON com CORS headers function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Busca múltiplos labels na coluna B e retorna mapa { chave: numeroLinha } function encontrarLinhas(sheet, labelsMap) { const maxRow = sheet.getLastRow(); const colB = sheet.getRange(1, 2, maxRow, 1).getValues(); const result = {}; Object.entries(labelsMap).forEach(([chave, texto]) => { for (let i = 0; i < colB.length; i++) { const cell = String(colB[i][0]).trim().toLowerCase(); if (cell === texto.toLowerCase()) { result[chave] = i + 1; // linha 1-indexed break; } } if (!result[chave]) result[chave] = null; }); return result; } // Busca um label na coluna B e retorna o número da linha function encontrarLinha(sheet, texto) { const maxRow = sheet.getLastRow(); const colB = sheet.getRange(1, 2, maxRow, 1).getValues(); for (let i = 0; i < colB.length; i++) { const cell = String(colB[i][0]).trim().toLowerCase(); if (cell === texto.toLowerCase()) return i + 1; } return null; } // Lê o valor de uma célula, retorna 0 se vazio ou erro function lerCelula(sheet, endereco) { try { const v = sheet.getRange(endereco).getValue(); return (v === "" || v === null || v === undefined) ? 0 : v; } catch(e) { return 0; } } // Lê as metas da aba Sonhos dinamicamente function lerSonhos(sheet) { const sonhos = []; const maxRow = sheet.getLastRow(); for (let r = 5; r <= maxRow; r++) { const nome = sheet.getRange(r, 2).getValue(); const meta = sheet.getRange(r, 4).getValue(); const atual = sheet.getRange(r, 5).getValue(); const prazo = sheet.getRange(r, 7).getValue(); // Ignora linhas vazias ou de subtotal if (!nome || typeof nome !== "string") continue; if (nome.toUpperCase().includes("SUBTOTAL") || nome.toUpperCase().includes("TOTAL")) continue; if (!meta || meta === 0) continue; sonhos.push({ nome: String(nome).trim(), meta: typeof meta === "number" ? meta : 0, atual: typeof atual === "number" ? atual : 0, prazo: prazo ? String(prazo) : "", pct: meta > 0 ? Math.round((atual / meta) * 100) : 0, }); } return sonhos; } // ════════════════════════════════════════════════════════════════════ // COMO TESTAR ANTES DE PUBLICAR // ════════════════════════════════════════════════════════════════════ // // 1. Logger.log("=== LINHAS ENCONTRADAS ==="); Logger.log(JSON.stringify(linhas)); Logger.log("=== MESES DETECTADOS (com ano) ==="); const SEQ_MESES = ["Jan","Fev","Mar","Abr","Mai","Jun","Jul","Ago","Set","Out","Nov","Dez"];
